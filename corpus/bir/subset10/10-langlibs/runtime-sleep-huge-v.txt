module $anon.. v 0.0.0;
woke  boolean;
sleeper(decimal) -> nil{
  bb0 {
    %2 = seconds;
    %3 = sleep(%2) -> bb1;
  }
  bb1 {
    lock-start "$anon/.:woke" GOTO bb2;
  }
  bb2 {
    PushScopeFrame 1
    %0 = ConstantLoad true
    woke = %0;
    PopScopeFrame
    lock-end "$anon/.:woke" GOTO bb3;
  }
  bb3 {
    return;
  }
}

    %2 = %1;

    %4 = %3;
    %5 = ConstantLoad 99999999999
    %6 = %5;

  }
  bb2 {
    %8 = %7;

    %10 = %9;
    %11 = sleep(%10) -> bb3;
  }
  bb3 {
    lock-start "$anon/.:woke" GOTO bb4;
  }
  bb4 {
    PushScopeFrame 1
    %0 = println(woke) -> bb5;
  }
  bb5 {
    PopScopeFrame
    lock-end "$anon/.:woke" GOTO bb6;
  }
  bb6 {
    return;
  }
}
